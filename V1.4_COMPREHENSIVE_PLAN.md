```cpp
char debugBuffer[DEBUG_BUFFER_SIZE];
bool debugPending = false;

void mqttDebug(const char* msg) {
  strncpy(debugBuffer, msg, DEBUG_BUFFER_SIZE - 1);
  debugPending = true;
}

// In loop():
void loop() {
  // ... existing code ...

  // Send debug if pending
  if (debugPending && (millis() - lastDebugSend > 100)) {
    StaticJsonDocument<256> doc;
    doc["type"] = "debug";
    doc["source"] = "UNO";
    doc["msg"] = debugBuffer;
    doc["ts"] = millis();

    serializeJson(doc, Serial);  // Send to ESP8266
    Serial.println();

    debugPending = false;
    lastDebugSend = millis();
  }
}
```

```cpp
// Handle debug messages from UNO
void handleUartFromUno() {
  String line = Serial.readStringUntil('\n');
  line.trim();

  StaticJsonDocument<512> doc;
  DeserializationError error = deserializeJson(doc, line);

  if (!error) {
    const char* type = doc["type"];

    if (strcmp(type, "debug") == 0) {
      // Forward to MQTT
      const char* source = doc["source"];
      const char* msg = doc["msg"];

      String payload = String(source) + ": " + String(msg);
      mqtt.publish("greenhouse/sys/debug", payload.c_str());
      return;
    }

    // ... handle other types ...
  }
}

// ESP8266 also has local debug:
void mqttDebug(const char* msg) {
  String payload = "ESP8266: " + String(msg);
  mqtt.publish("greenhouse/sys/debug", payload.c_str());
}
```

### 3. Sensor Filtering

#### **Median Filter for JSN-SR04T:**

```cpp
// UNO - Ultrasonic with median filter
#define ULTRASONIC_SAMPLES 5

float readUltrasonic() {
  float samples[ULTRASONIC_SAMPLES];

  // Collect samples
  for (int i = 0; i < ULTRASONIC_SAMPLES; i++) {
    digitalWrite(JSN_TRIG, LOW);
    delayMicroseconds(2);
    digitalWrite(JSN_TRIG, HIGH);
    delayMicroseconds(10);
    digitalWrite(JSN_TRIG, LOW);

    long duration = pulseIn(JSN_ECHO, HIGH, 30000);
    if (duration == 0) {
      samples[i] = -1;
    } else {
      samples[i] = duration * 0.034 / 2.0;
    }
    delay(10);  // Small delay between samples
  }

  // Sort and return median
  for (int i = 0; i < ULTRASONIC_SAMPLES - 1; i++) {
    for (int j = i + 1; j < ULTRASONIC_SAMPLES; j++) {
      if (samples[i] > samples[j]) {
        float temp = samples[i];
        samples[i] = samples[j];
        samples[j] = temp;
      }
    }
  }

  return samples[ULTRASONIC_SAMPLES / 2];  // Middle value
}
```

#### **Moving Average for Analog Sensors:**

```cpp
// UNO - Moving average filter
#define MA_SIZE 10

class MovingAverage {
private:
  int buffer[MA_SIZE];
  int index;
  long sum;
  int count;

public:
  MovingAverage() : index(0), sum(0), count(0) {}

  int add(int value) {
    if (count < MA_SIZE) {
      count++;
    } else {
      sum -= buffer[index];
    }

    buffer[index] = value;
    sum += value;
    index = (index + 1) % MA_SIZE;

    return sum / count;
  }
};

// Global filters
MovingAverage soilFilter;
MovingAverage flameFilter;
MovingAverage soundFilter;
MovingAverage mq3Filter;

// In readAllSensors():
void readAllSensors() {
  // Soil moisture with filter
  int soilRaw = analogRead(ANALOG_SOIL);
  int soilFiltered = soilFilter.add(soilRaw);
  sensors.soil_pct = map(soilFiltered, 0, 1023, 0, 100);

  // MQ-3 with filter
  int mq3Raw = analogRead(ANALOG_MQ3);
  sensors.mq3 = mq3Filter.add(mq3Raw);

  // Flame with filter
  int flameRaw = analogRead(ANALOG_FLAME);
  sensors.flame = flameFilter.add(flameRaw);

  // Sound with filter
  int soundRaw = analogRead(ANALOG_SOUND);
  sensors.sound = soundFilter.add(soundRaw);

  // BH1750 - already stable, no filter needed
  sensors.light_lux = lightMeter.readLightLevel();

  // Ultrasonic with median filter
  sensors.distance_cm = readUltrasonic();
}
```

```cpp
void publishSensorData() {
  if (!mqtt.connected()) return;

  unsigned long timestamp = millis();

  // Publish with timestamp and unit
  publishSensor("temperature", sensorData.temp_c, "°C", timestamp);
  publishSensor("humidity", sensorData.hum_pct, "%", timestamp);
  publishSensor("soilMoisture", sensorData.soil_pct, "%", timestamp);
  publishSensor("lightIntensity", sensorData.light_lux, "lux", timestamp);
  publishSensor("waterTankLevel", sensorData.distance_cm, "cm", timestamp);
  publishSensor("gasMQ3", sensorData.mq3 * 5.0 / 1023.0, "V", timestamp);
  publishSensor("flameAnalog", (float)sensorData.flame, "raw", timestamp);
  publishSensor("soundLevel", (float)sensorData.sound, "raw", timestamp);
}

void publishSensor(const char* name, float value, const char* unit, unsigned long ts) {
  StaticJsonDocument<128> doc;
  doc["v"] = value;
  doc["u"] = unit;
  doc["t"] = ts;

  String json;
  serializeJson(doc, json);

  String topic = "greenhouse/data/";
  topic += name;
  mqtt.publish(topic.c_str(), json.c_str());
}
```

### 5. Default Thresholds

#### **ESP8266 Implementation:**

```cpp
// Default thresholds
struct Thresholds {
  float temp_max = 35.0;
  float temp_min = 15.0;
  float hum_max = 80.0;
  float hum_min = 40.0;
  float soil_min = 30.0;
  int flame_threshold = 800;
  int sound_threshold = 700;
};

Thresholds thresholds;
bool useDefaults = true;

// Load from LittleFS or use defaults
void loadThresholds() {
  File file = LittleFS.open("/thresholds.json", "r");
  if (file) {
    StaticJsonDocument<256> doc;
    DeserializationError error = deserializeJson(doc, file);

    if (!error) {
      thresholds.temp_max = doc["temp_max"] | 35.0;
      thresholds.temp_min = doc["temp_min"] | 15.0;
      thresholds.hum_max = doc["hum_max"] | 80.0;
      thresholds.hum_min = doc["hum_min"] | 40.0;
      thresholds.soil_min = doc["soil_min"] | 30.0;
      thresholds.flame_threshold = doc["flame_threshold"] | 800;
      thresholds.sound_threshold = doc["sound_threshold"] | 700;
      useDefaults = false;
      mqttDebug("Thresholds loaded from file");
    }
    file.close();
  } else {
    mqttDebug("Using default thresholds");
  }

  // Publish current thresholds
  publishThresholds();
}

// Auto control based on thresholds
void autoControl() {
  if (currentMode != "AUTO") return;

  // Temperature control
  if (sensorData.temp_c > thresholds.temp_max) {
    // Turn on fan
    if (deviceState.fan != "ON") {
      handleDeviceCommand("fan", "ON");
      mqttDebug("Auto: Fan ON (temp high)");
    }
  } else if (sensorData.temp_c < thresholds.temp_min) {
    // Turn off fan
    if (deviceState.fan != "OFF") {
      handleDeviceCommand("fan", "OFF");
      mqttDebug("Auto: Fan OFF (temp normal)");
    }
  }

  // Soil moisture control
  if (sensorData.soil_pct < thresholds.soil_min) {
    // Turn on pump
    if (deviceState.pump != "ON") {
      handleDeviceCommand("pump", "ON");
      mqttDebug("Auto: Pump ON (soil dry)");
    }
  }

  // Flame detection (always active)
  if (sensorData.flame > thresholds.flame_threshold) {
    mqttDebug("ALERT: Flame detected!");
    // Emergency actions
  }
}
```

### 6. Optimize NeoPixel show()

#### **UNO Implementation:**

```cpp
// Track previous LED state
struct LedState {
  uint8_t r, g, b;
  bool needsUpdate;
};

LedState lastLedState = {0, 0, 0, false};

void setLedColor(uint8_t r, uint8_t g, uint8_t b) {
  // Only update if color actually changed
  if (r != lastLedState.r || g != lastLedState.g || b != lastLedState.b) {
    lastLedState.r = r;
    lastLedState.g = g;
    lastLedState.b = b;
    lastLedState.needsUpdate = true;
  }
}

void updateLeds() {
  if (lastLedState.needsUpdate) {
    for (int i = 0; i < WS2812_COUNT; i++) {
      strip.setPixelColor(i, strip.Color(lastLedState.r, lastLedState.g, lastLedState.b));
    }
    strip.show();  // Only show when actually changed
    lastLedState.needsUpdate = false;
  }
}

// In loop():
void loop() {
  // ... existing code ...

  // Update LEDs only when needed
  updateLeds();
}
```

## ⚠️ MIGRATION NOTES

### **Hardware Changes Required:**
```
OLD Wiring:
UNO D8 (RX) ←→ ESP8266 GPIO14 (TX)
UNO D9 (TX) ←→ ESP8266 GPIO12 (RX)

NEW Wiring:
UNO D0 (RX) ←→ ESP8266 GPIO3 (RX0/TX)
UNO D1 (TX) ←→ ESP8266 GPIO1 (TX0/RX)
```

**Level Shifting:**
- UNO: 5V logic
- ESP8266: 3.3V logic
- **REQUIRED:** Voltage divider or level shifter for UNO TX → ESP8266 RX
- ESP8266 TX → UNO RX: Usually OK (3.3V recognized as HIGH by 5V logic)

### **Software Changes:**
- Remove AltSoftSerial library from UNO
- Update pin definitions
- Replace all Serial.print() debug with mqttDebug()
- Flash both UNO and ESP8266 with v1.4 firmware
